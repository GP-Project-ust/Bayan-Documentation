% REST + SSE conversation sequence — Chapter 4, Section 4.6.1

%==============================================================================
% API-boundary sequence diagram for one chat request: the tenant user asks
% an Arabic question in the Next.js chat UI, which POSTs it to the FastAPI
% gateway (JWT header); the gateway enforces the security checks of 4.5
% (token, membership, capability) then invokes the agents graph (4.2);
% processing steps stream back over the SSE channel as status/trace events
% feeding the "thinking now" strip, then the unified payload arrives as
% sources + final events, and the gateway closes by recording the audit
% trail with the request id (4.5.3).
%
% Distinct from the chapter-3 agent-sequence diagram (17 inter-agent
% messages): this figure documents the HTTP/SSE contract only — 11
% numbered messages, 4 phase bands, 5 lifelines right-to-left.
% Solid arrows = HTTP calls, dashed = returns/rendering, kauaccent dashed
% = SSE event frames. Arabic labels in \textarabic{}, Latin in \EN{}.
% Anchored in the reference implementation (backend/app): the route
% POST /api/v1/chat/query, the toolSteps trace fields, and the unified
% payload sqlResult/chartConfig/answer.
%==============================================================================
\resizebox{0.97\textwidth}{!}{%
\begin{tikzpicture}[
  >=stealth,
  font=\small,
  x=1cm, y=1cm,
  head/.style={
    rectangle, rounded corners=4pt, draw=kauprimary, fill=kauprimary!10,
    text=kaudark, align=center, inner sep=4pt,
    font=\scriptsize\bfseries, minimum width=2.35cm, minimum height=0.82cm
  },
  call/.style={->, >=stealth, line width=0.75pt, draw=kaudark!80},
  ret/.style={->, >=stealth, line width=0.6pt, draw=kausecondary!90!black, dashed},
  ssee/.style={->, >=stealth, line width=0.8pt, draw=kauaccent!95!kaudark, dashed},
  selfm/.style={->, >=stealth, line width=0.6pt, draw=kaudark!75},
  lifeline/.style={draw=kaudark!45, dashed, line width=0.5pt},
  phase/.style={font=\tiny\bfseries, text=kauprimary!85!black},
  msgl/.style={font=\tiny, text=kaudark, fill=white, inner sep=0.8pt, align=center},
  sselbl/.style={font=\tiny\bfseries, text=kauaccent!85!kaudark, fill=white, inner sep=0.8pt},
  actbar/.style={fill=kausecondary!30, draw=kausecondary!80!black, line width=0.4pt},
  num/.style={
    circle, fill=kauprimary!85, text=white, font=\tiny\bfseries,
    inner sep=0.6pt, minimum size=3.2mm, text width=2.8mm, align=center,
  },
  framenote/.style={
    rectangle, rounded corners=3pt, draw=kaudark!55, fill=white,
    inner sep=4pt, align=left, font=\tiny, text=kaudark,
  },
]

  %==========================================================================
  % Lifeline x-positions (right -> left = RTL reading order)
  %==========================================================================
  \def\ux{8.6}     % مستخدم المؤسسة
  \def\uix{5.9}    % واجهة المحادثة (Next.js)
  \def\apx{3.2}    % البوابة الخلفية (FastAPI)
  \def\agx{0.5}    % منظومة الوكلاء (LangGraph)
  \def\aux{-2.2}   % سجل التدقيق

  % ---- Phase bands (alternating tints, drawn first) ----
  \fill[kauprimary!4]  (9.9,-1.05)  rectangle (-4.9,-4.40);
  \fill[kausecondary!5] (9.9,-4.40)  rectangle (-4.9,-8.00);
  \fill[kauprimary!4]  (9.9,-8.00)  rectangle (-4.9,-11.00);
  \fill[kausecondary!5] (9.9,-11.00) rectangle (-4.9,-12.95);

  % ---- Phase separators + labels (right margin = first read) ----
  \foreach \yy in {-4.40,-8.00,-11.00}{%
    \draw[draw=dividersilver, dotted, line width=0.5pt] (9.9,\yy) -- (-4.9,\yy);
  }
  \node[phase] at (9.15,-1.32)  {\textarabic{المرحلة 1: الطلب والتحقق}};
  \node[phase] at (9.15,-4.66)  {\textarabic{المرحلة 2: التنفيذ والبث المتدرج}};
  \node[phase] at (9.15,-8.26)  {\textarabic{المرحلة 3: الإجابة النهائية}};
  \node[phase] at (9.15,-11.26) {\textarabic{المرحلة 4: التوثيق}};

  \foreach \xx in {8.6,5.9,3.2,0.5,-2.2}{%

  }

  % ---- Header boxes ----
  \node[head] at (\ux,0)  {\textarabic{مستخدم المؤسسة}\\[-1pt]{\tiny\EN{(Tenant User)}}};
  \node[head] at (\uix,0) {\textarabic{واجهة المحادثة}\\[-1pt]{\tiny\EN{(Next.js)}}};
  \node[head, draw=kauprimary, fill=kauprimary!16, line width=0.9pt] at (\apx,0)
    {\textarabic{البوابة الخلفية}\\[-1pt]{\tiny\EN{(FastAPI)}}};

    {\textarabic{سجل التدقيق}\\[-1pt]{\tiny\EN{(Audit Logs)}}};

  % ---- Activation bars ----
  \fill[actbar] (\uix-0.07,-2.20) rectangle (\uix+0.07,-10.75);  % UI
  \fill[actbar] (\apx-0.07,-2.30)  rectangle (\apx+0.07,-12.05); % Gateway
  \fill[actbar] (\agx-0.07,-3.95)  rectangle (\agx+0.07,-8.95);  % Agents
  \fill[actbar] (\aux-0.07,-11.80) rectangle (\aux+0.07,-12.30); % Audit

  %==========================================================================
  % PHASE 1 — الطلب والتحقق
  %==========================================================================
  \draw[call] (\ux,-1.70) -- (\uix,-1.70)
    node[msgl, pos=0.5, above=1.5pt] {\textarabic{يرسل السؤال العربي}};
  \node[num] at (7.55,-1.70) {1};

  \draw[call] (\uix,-2.55) -- (\apx,-2.55)
    node[msgl, pos=0.5, above=1.5pt] {\EN{POST /api/v1/chat/query}};
  \node[msgl] at (4.55,-2.95) {\textarabic{مع رأس الرمز}~\EN{JWT}};
  \node[num] at (4.90,-2.55) {2};

  \draw[selfm] (\apx+0.07,-3.30) -- (\apx+0.85,-3.30)
    -- (\apx+0.85,-3.95) -- (\apx+0.07,-3.95)
    node[msgl, pos=0.5, right=2pt] {\textarabic{فحص الرمز والعضوية والقدرة (4.5)}};

  \draw[call] (\apx,-4.15) -- (\agx,-4.15)

  \draw[ssee] (\apx,-6.15) -- (\uix,-6.15)
    node[msgl, pos=0.45, above=1.5pt] {\textarabic{حدث}~\EN{status}~\textarabic{و}~\EN{trace}};
  \node[sselbl] at (4.55,-6.62) {\textarabic{«جارٍ توليد الاستعلام...» — بث}~\EN{SSE}};

  \draw[ret] (\uix,-7.20) -- (\ux,-7.20)
    node[msgl, pos=0.5, above=1.5pt] {\textarabic{شريط «يفكر الآن» ورسائل المراحل}};
  \node[num] at (7.55,-7.20) {7};

  %==========================================================================
  % PHASE 3 — الإجابة النهائية
  %==========================================================================

  \node[num] at (1.35,-8.75) {8};

  \draw[ssee] (\apx,-9.70) -- (\uix,-9.70)
    node[msgl, pos=0.5, above=1.5pt] {\textarabic{حدث}~\EN{sources}~\textarabic{ثم}~\EN{final}};
  \node[num] at (4.90,-9.70) {9};

  \draw[ret] (\uix,-10.55) -- (\ux,-10.55)
    node[msgl, pos=0.5, above=1.5pt]
      {\textarabic{الإجابة كاملة بالمصادر وزر «تصدير التقرير»}};
  \node[num] at (7.55,-10.55) {10};

  %==========================================================================
  % PHASE 4 — التوثيق
  %==========================================================================
  \draw[call] (\apx,-12.00) -- (\aux,-12.00)
