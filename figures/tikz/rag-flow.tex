\begin{tikzpicture}[
  >=stealth,
  % Process node
  process/.style={
    rectangle, rounded corners=4pt,
    draw=kauprimary, fill=kaulight, text=kaudark,
    font=\small, align=center,
    minimum width=4.5cm, minimum height=0.8cm,
    inner sep=4pt,
  },
  % Phase 1 (Retrieval) - blue
  phase1/.style={
    rectangle, rounded corners=4pt,
    draw=kauprimary, fill=kauprimary!10, text=kaudark,
    font=\small, align=center,
    minimum width=4.5cm, minimum height=0.8cm,
    inner sep=4pt,
  },
  % Phase 2 (Generation) - red
  phase2/.style={
    rectangle, rounded corners=4pt,
    draw=kauaccent, fill=kauaccent!10, text=kaudark,
    font=\small, align=center,
    minimum width=4.5cm, minimum height=0.8cm,
    inner sep=4pt,
  },
  % Data store
  datastore/.style={
    cylinder, shape border rotate=90, aspect=0.3,
    draw=kausecondary, fill=kausecondary!12, text=kaudark,
    font=\footnotesize, align=center,
    minimum width=2.8cm, minimum height=1cm,
    inner sep=2pt,
  },
  % Output node
  output/.style={
    rectangle, rounded corners=4pt,
    draw=kaudark, fill=kaudark!8, text=kaudark,
    font=\small\bfseries, align=center,
    minimum width=5.5cm, minimum height=0.9cm,
    inner sep=6pt,
  },
  % Arrow styles
  arrow/.style={->, >=stealth, line width=1pt, draw=kauprimary},
  arrow2/.style={->, >=stealth, line width=1pt, draw=kauaccent},
  arrowDash/.style={->, >=stealth, line width=0.8pt, draw=kausecondary, dashed},
  % Label on arrow
  arlabel/.style={
    font=\scriptsize, text=kaudark,
    fill=white, inner sep=1pt,
  },
]

  %------------------------------------------------------------------------
  % Main vertical flow (centered)
  %------------------------------------------------------------------------

  % 1. User Question
  \node[process] (question) at (0, 0)
    {\RL{سؤال المستخدم}\\\footnotesize\EN{User Question}};

  % 2. RAG Agent
  \node[phase1, below=0.8cm of question] (ragagent)
    {\RL{وكيل \EN{RAG}}};

  % 3. Embedding Model
  \node[phase1, below=0.8cm of ragagent] (embed)
    {\RL{نموذج التضمين}\\\footnotesize\EN{multilingual-e5-large}};

  % 4. Retrieved Context
  \node[phase1, below=0.8cm of embed] (context)
    {\RL{المستندات المسترجعة}\\\footnotesize\EN{Top-K Chunks}};

  % 5. LLM
  \node[phase2, below=0.8cm of context] (llm)
    {\RL{النموذج اللغوي}\\\footnotesize\EN{DeepSeek / GPT-4o}};

  % 6. Augmented Prompt
  \node[phase2, below=0.8cm of llm] (prompt)
    {\RL{توليد الإجابة المعززة}};

  % 7. Final Answer
  \node[output, below=0.8cm of prompt] (answer)
    {\RL{الإجابة + الاستشهادات + مسار التنفيذ}};

  %------------------------------------------------------------------------
  % Side elements (data stores)
  %------------------------------------------------------------------------

  % Document Store (left of embedding)
  \node[datastore, left=2.2cm of embed] (docs)
    {\RL{مستندات \EN{Odoo}}};

  % Vector Database (right of embedding)
  \node[datastore, right=2.2cm of embed] (qdrant)
    {\RL{\EN{Qdrant}}\\\RL{(متجهات)}};

  %------------------------------------------------------------------------
  % Arrows (main flow)
  %------------------------------------------------------------------------

  \draw[arrow] (question) -- (ragagent);
  \draw[arrow] (ragagent) -- (embed)
    node[arlabel, midway, right=2pt] {\EN{1. Embed}};
  \draw[arrow] (embed) -- (context)
    node[arlabel, midway, right=2pt] {\EN{3. Retrieve}};
  \draw[arrow2] (context) -- (llm)
    node[arlabel, midway, right=2pt] {\EN{4. Context}};
  \draw[arrow2] (llm) -- (prompt)
    node[arlabel, midway, right=2pt] {\EN{5. Generate}};
  \draw[arrow2] (prompt) -- (answer)
    node[arlabel, midway, right=2pt] {\EN{6. Output}};

  %------------------------------------------------------------------------
  % Side arrows (search & index)
  %------------------------------------------------------------------------

  % Embed → Qdrant (search)
  \draw[arrow] (embed.east) -- (qdrant.west)
    node[arlabel, midway, above] {\EN{2. Search}};

  % Docs → Qdrant (indexing, dashed)
  \draw[arrowDash] (docs.east) -- (qdrant.west)
    node[arlabel, midway, below] {\EN{Index}};

  %------------------------------------------------------------------------
  % Phase labels (far right)
  %------------------------------------------------------------------------
  \node[font=\small\bfseries\color{kauprimary}, rotate=90]
    at (3.5, -1.8) {\RL{الاسترجاع}};

  \node[font=\small\bfseries\color{kauaccent}, rotate=90]
    at (3.5, -4.2) {\RL{التوليد}};

\end{tikzpicture}
