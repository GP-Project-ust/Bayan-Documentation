% RAG Pipeline Flow Diagram for Bayan System (v3 - RTL reading order)
%==============================================================================
% RTL fix (2026-09):
%   1) Every Arabic label is wrapped in \textarabic{} so it typesets with
%      correct RTL word order inside TikZ's LTR nodes (same proven fix as
%      Figures 3-2/3-3/3-4 and the Gantt/use-case diagrams).
%   2) The side data stores now sit in a stacked sidebar on the LEFT (read
%      last in RTL order): Odoo documents (top) are indexed downward into
%      the Qdrant vector store, and the search arrow points leftward from
%      the embedding node. This also removes the original layout defect
%      where the dashed Index arrow crossed straight through the embed box.
%   3) The main pipeline remains vertical (top-to-bottom), which is
%      direction-neutral for Arabic readers; phase labels annotate from the
%      right (the first-read side in an RTL document).
%==============================================================================

\begin{tikzpicture}[
  >=stealth,
  % Process node
  process/.style={
    rectangle, rounded corners=4pt,
    draw=kauprimary, fill=kaulight, text=kaudark,
    font=\small, align=center,
    minimum width=4.5cm, minimum height=0.8cm,
    inner sep=4pt,
  },
  % Phase 1 (Retrieval) - blue
  phase1/.style={
    rectangle, rounded corners=4pt,
    draw=kauprimary, fill=kauprimary!10, text=kaudark,
    font=\small, align=center,
    minimum width=4.5cm, minimum height=0.8cm,
    inner sep=4pt,
  },
  % Phase 2 (Generation) - red
  phase2/.style={
    rectangle, rounded corners=4pt,
    draw=kauaccent, fill=kauaccent!10, text=kaudark,
    font=\small, align=center,
    minimum width=4.5cm, minimum height=0.8cm,
    inner sep=4pt,
  },
  % Data store
  datastore/.style={
    cylinder, shape border rotate=90, aspect=0.3,
    draw=kausecondary, fill=kausecondary!12, text=kaudark,
    font=\footnotesize, align=center,
    minimum width=2.8cm, minimum height=1cm,
    inner sep=2pt,
  },
  % Output node
  output/.style={
    rectangle, rounded corners=4pt,
    draw=kaudark, fill=kaudark!8, text=kaudark,
    font=\small\bfseries, align=center,
    minimum width=5.5cm, minimum height=0.9cm,
    inner sep=6pt,
  },
  % Arrow styles
  arrow/.style={->, >=stealth, line width=1pt, draw=kauprimary},
  arrow2/.style={->, >=stealth, line width=1pt, draw=kauaccent},
  arrowDash/.style={->, >=stealth, line width=0.8pt, draw=kausecondary, dashed},
  % Label on arrow
  arlabel/.style={
    font=\scriptsize, text=kaudark,
    fill=white, inner sep=1pt,
  },
]

  %------------------------------------------------------------------------
  % Main vertical flow (centered)
  %------------------------------------------------------------------------

  % 1. User Question
  \node[process] (question) at (0, 0)
    {\textarabic{سؤال المستخدم}\\\footnotesize\EN{User Question}};

  % 2. RAG Agent
  \node[phase1, below=0.8cm of question] (ragagent)
    {\textarabic{وكيل \EN{RAG}}};

  % 3. Embedding Model
  \node[phase1, below=0.8cm of ragagent] (embed)
    {\textarabic{نموذج التضمين}\\\footnotesize\EN{multilingual-e5-large}};

  % 4. Retrieved Context
  \node[phase1, below=0.8cm of embed] (context)
    {\textarabic{المستندات المسترجعة}\\\footnotesize\EN{Top-K Chunks}};

  % 5. LLM
  \node[phase2, below=0.8cm of context] (llm)
    {\textarabic{النموذج اللغوي}\\\footnotesize\EN{DeepSeek / GPT-4o}};

  % 6. Augmented Prompt
  \node[phase2, below=0.8cm of llm] (prompt)
    {\textarabic{توليد الإجابة المعززة}};

  % 7. Final Answer
  \node[output, below=0.8cm of prompt] (answer)
    {\textarabic{الإجابة + الاستشهادات + مسار التنفيذ}};

  %------------------------------------------------------------------------
  % Side data stores (stacked sidebar on the LEFT — read last in RTL).
  % Qdrant sits beside the embedding node; Odoo documents sit above it so
  % the indexing arrow flows downward into the vector store.
  %------------------------------------------------------------------------

  % Vector Database (left of embedding)
  \node[datastore, left=2.2cm of embed] (qdrant)
    {\EN{Qdrant}\\\textarabic{(متجهات)}};

  % Document Store (above Qdrant)
  \node[datastore, above=0.35cm of qdrant] (docs)
    {\textarabic{مستندات \EN{Odoo}}};

  %------------------------------------------------------------------------
  % Arrows (main flow)
  %------------------------------------------------------------------------

  \draw[arrow] (question) -- (ragagent);
  \draw[arrow] (ragagent) -- (embed)
    node[arlabel, midway, right=2pt] {\EN{1. Embed}};
  \draw[arrow] (embed) -- (context)
    node[arlabel, midway, right=2pt] {\EN{3. Retrieve}};
  \draw[arrow2] (context) -- (llm)
    node[arlabel, midway, right=2pt] {\EN{4. Context}};
  \draw[arrow2] (llm) -- (prompt)
    node[arlabel, midway, right=2pt] {\EN{5. Generate}};
  \draw[arrow2] (prompt) -- (answer)
    node[arlabel, midway, right=2pt] {\EN{6. Output}};

  %------------------------------------------------------------------------
  % Side arrows (search & index) — no crossings
  %------------------------------------------------------------------------

  % Embed → Qdrant (search, pointing leftward = RTL natural)
  \draw[arrow] (embed.west) -- (qdrant.east)
    node[arlabel, midway, above] {\EN{2. Search}};

  % Docs → Qdrant (indexing, dashed, flowing downward into the store)
  \draw[arrowDash] (docs.south) -- (qdrant.north)
    node[arlabel, midway, right=2pt] {\EN{Index}};

  %------------------------------------------------------------------------
  % Phase labels (right side — first-read side in an RTL document)
  %------------------------------------------------------------------------
  \node[font=\small\bfseries\color{kauprimary}, rotate=90]
    at (3.7, -1.8) {\textarabic{الاسترجاع}};

  \node[font=\small\bfseries\color{kauaccent}, rotate=90]
    at (3.7, -4.2) {\textarabic{التوليد}};

\end{tikzpicture}
