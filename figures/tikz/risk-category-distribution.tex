\resizebox{0.85\textwidth}{!}{%
\begin{tikzpicture}[

  >=stealth,
  title/.style={font=\large\bfseries, kauprimary},
  axistitle/.style={font=\small\bfseries, kauprimary},
  catlabel/.style={font=\scriptsize, kaudark, align=center, anchor=north,

}

% تشغيلية = 3

% اقتصادية = 2

% قانونية وامتثال = 3
