%==============================================================================
% figures/tikz/usecase-sequence-vA.tex
% Comprehensive Sequence Diagram (Version A) — All 8 Use Cases (UC-01..UC-08)
% Bayan (بيان) — Arabic AI Assistant System
%
% Design summary:
%   • 9 lifelines (RTL ordering: User → Frontend → Supervisor →
%     SQL/RAG/Analyst agents → PostgreSQL/Qdrant → Audit Log)
%   • 8 grouped UC blocks with distinct background tints per group
%   • 3–5 KEY messages per UC + 1 universal audit-logging arrow at the
%     bottom of each block (UC-07 uses a self-loop since the audit log
%     is itself the target of the flow).
%   • Header & footer boxes repeat lifeline labels for readability
%   • Wrapped in \resizebox{\textwidth}{!}{...} for single A4 page fit
%
% Dependencies (defined in main.tex — do NOT build standalone):
%   – Colors : kauprimary, kausecondary, kauaccent, kaulight, kaudark,
%              dividerdim, dividersilver
%   – Macro  : \EN{...} = \textenglish{\englishfont #1}
%   – TikZ   : libraries calc, shapes.geometric, positioning (loaded in main)
%
% Actor mapping (the "المستخدم المؤسسي" lifeline hosts multiple actors):
%   UC-01..UC-04 → Organizational User (موظف / مدير المؤسسة)
%   UC-05..UC-07 → System Admin       (مسؤول النظام)
%   UC-08        → Onboarding Operator (مشغل التهيئة)
%==============================================================================

\resizebox{\textwidth}{!}{%
\begin{tikzpicture}[
  font=\small,
  header/.style={
    draw=kauprimary!60, fill=kauprimary!12, line width=0.9pt,
    rounded corners=3pt, align=center, inner sep=4pt,
    text=kaudark, font=\bfseries\small,
    minimum width=1.7cm, minimum height=0.7cm
  },
  lifeline/.style={draw=dividerdim, line width=0.6pt, dashed},
  msg/.style={-stealth, line width=0.7pt, draw=kaudark},
  reply/.style={-stealth, line width=0.7pt, draw=kaudark, dashed},
  mlabel/.style={font=\small, text=kaudark, fill=white, inner sep=1.5pt},
  ucblock/.style={draw=kauprimary!50, line width=0.8pt, rounded corners=2pt,
                  fill=kauprimary!4},
  uclabel/.style={fill=kauprimary!20, font=\bfseries\small, text=kaudark,
                  inner sep=3pt, rounded corners=2pt},
  selfloop/.style={-stealth, line width=0.7pt, draw=kaudark}
]

  %==========================================================================
  %  LIFELINE X-POSITIONS  (9 columns, 2.3 units apart — tightened to reduce
  %  the scaling factor so the \small font survives \resizebox better)
  %==========================================================================
  \def\xUser{0}
  \def\xFront{2.3}
  \def\xSup{4.6}
  \def\xSql{6.9}
  \def\xRag{9.2}
  \def\xAna{11.5}
  \def\xPg{13.8}
  \def\xQd{16.1}
  \def\xAudit{18.4}

  %==========================================================================
  %  TOP HEADER BOXES
  %==========================================================================
  \node[header] (hUser)  at (\xUser,  0) {المستخدم المؤسسي\\\EN{(Actor)}};
  \node[header] (hFront) at (\xFront, 0) {الواجهة الأمامية\\\EN{(Frontend)}};
  \node[header] (hSup)   at (\xSup,   0) {الوكيل المشرف\\\EN{(Supervisor)}};
  \node[header] (hSql)   at (\xSql,   0) {وكيل \EN{SQL}};
  \node[header] (hRag)   at (\xRag,   0) {وكيل \EN{RAG}};
  \node[header] (hAna)   at (\xAna,   0) {وكيل المحلل\\\EN{(Analyst)}};
  \node[header] (hPg)    at (\xPg,    0) {\EN{PostgreSQL}};
  \node[header] (hQd)    at (\xQd,    0) {\EN{Qdrant}};
  \node[header] (hAudit) at (\xAudit, 0) {سجل التدقيق\\\EN{(Audit Log)}};

  %==========================================================================
  %  LIFELINES  (vertical dashed lines from header to just above footer)
  %==========================================================================
  \def\yEnd{-23.6}
  \foreach \x in {\xUser, \xFront, \xSup, \xSql, \xRag, \xAna, \xPg, \xQd, \xAudit} {
    \draw[lifeline] (\x, -0.45) -- (\x, \yEnd);
  }

  %==========================================================================
  %  MESSAGE MACROS
  %    #1 = source x, #2 = target x, #3 = y-coordinate, #4 = label
  %==========================================================================
  \newcommand{\smsg}[4]{\draw[msg]   (#1, #3) -- (#2, #3) node[midway, mlabel, above] {#4};}
  \newcommand{\rmsg}[4]{\draw[reply] (#1, #3) -- (#2, #3) node[midway, mlabel, above] {#4};}

  %==========================================================================
  %  UC-01: طرح استعلام عن البيانات   (block: y = -1.0 .. -4.0)
  %    User → Frontend → Supervisor → SQL Agent → PostgreSQL → (return)
  %==========================================================================
  \draw[ucblock, fill=kauprimary!5] (-0.9, -1.0) rectangle (19.3, -4.0);
  \node[uclabel, anchor=north west] at (-0.85, -1.05)
    {\EN{UC-01:} طرح استعلام عن البيانات};

  \smsg{\xUser}{\xFront}{-1.65}{يكتب سؤالاً عن البيانات}
  \smsg{\xSup}{\xSql}{-2.20}{توجيه (\EN{NL2SQL} + \EN{Smart Views})}
  \smsg{\xSql}{\xPg}{-2.75}{تنفيذ \EN{SQL} (مع \EN{RBAC})}
  \rmsg{\xSup}{\xUser}{-3.30}{إرجاع النتيجة المنسقة}
  \smsg{\xSql}{\xAudit}{-3.85}{تسجيل التدقيق}

  %==========================================================================
  %  UC-02: البحث في المستندات   (block: y = -4.1 .. -7.1)
  %    User → Frontend → Supervisor → RAG Agent → Qdrant → (return)
  %==========================================================================
  \draw[ucblock, fill=kauprimary!7] (-0.9, -4.1) rectangle (19.3, -7.1);
  \node[uclabel, anchor=north west] at (-0.85, -4.15)
    {\EN{UC-02:} البحث في المستندات};

  \smsg{\xUser}{\xFront}{-4.75}{يكتب سؤالاً وثائقياً}
  \smsg{\xSup}{\xRag}{-5.30}{توجيه (\EN{Query Embedding})}
  \smsg{\xRag}{\xQd}{-5.85}{استرجاع \EN{Top-K}}
  \rmsg{\xSup}{\xUser}{-6.40}{إرجاع الإجابة + المصادر}
  \smsg{\xRag}{\xAudit}{-6.95}{تسجيل التدقيق}

  %==========================================================================
  %  UC-03: توليد تقرير مخصص   (block: y = -7.2 .. -10.2)
  %    User → Frontend → Supervisor → Analyst ↔ SQL → (return + export)
  %==========================================================================
  \draw[ucblock, fill=kauprimary!9] (-0.9, -7.2) rectangle (19.3, -10.2);
  \node[uclabel, anchor=north west] at (-0.85, -7.25)
    {\EN{UC-03:} توليد تقرير مخصص};

  \smsg{\xUser}{\xFront}{-7.85}{يطلب تقريراً تحليلياً}
  \smsg{\xSup}{\xAna}{-8.40}{توجيه الطلب}
  \smsg{\xAna}{\xSql}{-8.95}{جلب وتحليل البيانات}
  \rmsg{\xSup}{\xUser}{-9.50}{إرجاع التقرير + تصدير}
  \smsg{\xAna}{\xAudit}{-10.05}{تسجيل التدقيق}

  %==========================================================================
  %  UC-04: عرض الإشعارات   (block: y = -10.3 .. -12.3)
  %    User → Frontend → (Notifications list)
  %==========================================================================
  \draw[ucblock, fill=kaulight] (-0.9, -10.3) rectangle (19.3, -12.3);
  \node[uclabel, anchor=north west] at (-0.85, -10.35)
    {\EN{UC-04:} عرض الإشعارات};

  \smsg{\xUser}{\xFront}{-10.95}{يطلب عرض الإشعارات}
  \rmsg{\xFront}{\xUser}{-11.50}{عرض قائمة الإشعارات}
  \smsg{\xFront}{\xAudit}{-12.05}{تسجيل التدقيق}

  %==========================================================================
  %  UC-05: إدارة المستخدمين والأدوار   (block: y = -12.4 .. -15.0)
  %    System Admin → Frontend → PostgreSQL (RBAC update)
  %==========================================================================
  \draw[ucblock, fill=kausecondary!7] (-0.9, -12.4) rectangle (19.3, -15.0);
  \node[uclabel, anchor=north west] at (-0.85, -12.45)
    {\EN{UC-05:} إدارة المستخدمين والأدوار};

  \smsg{\xUser}{\xFront}{-13.05}{إنشاء/تعديل مستخدم أو دور}
  \smsg{\xFront}{\xPg}{-13.60}{تحديث جدول \EN{RBAC}}
  \rmsg{\xPg}{\xFront}{-14.15}{تأكيد التحديث}
  \smsg{\xFront}{\xAudit}{-14.70}{تسجيل التدقيق}

  %==========================================================================
  %  UC-06: إدارة مصادر البيانات   (block: y = -15.1 .. -17.7)
  %    System Admin → Frontend → Smart Views → PostgreSQL schema update
  %==========================================================================
  \draw[ucblock, fill=kausecondary!9] (-0.9, -15.1) rectangle (19.3, -17.7);
  \node[uclabel, anchor=north west] at (-0.85, -15.15)
    {\EN{UC-06:} إدارة مصادر البيانات};

  \smsg{\xUser}{\xFront}{-15.75}{إضافة/تعديل مصدر بيانات}
  \smsg{\xFront}{\xPg}{-16.30}{تحديث مخطط \EN{Smart Views}}
  \rmsg{\xPg}{\xFront}{-16.85}{تأكيد + جاهزية}
  \smsg{\xFront}{\xAudit}{-17.40}{تسجيل التدقيق}

  %==========================================================================
  %  UC-07: مراجعة سجل التدقيق   (block: y = -17.8 .. -20.4)
  %    System Admin → Frontend → Audit Log (filter / export CSV)
  %    The admin's access to the audit log is itself logged (self-loop).
  %==========================================================================
  \draw[ucblock, fill=dividersilver!15] (-0.9, -17.8) rectangle (19.3, -20.4);
  \node[uclabel, anchor=north west] at (-0.85, -17.85)
    {\EN{UC-07:} مراجعة سجل التدقيق};

  \smsg{\xUser}{\xFront}{-18.45}{يطلب مراجعة السجل + تصفية}
  \smsg{\xFront}{\xAudit}{-19.00}{استعلام + تصفية}
  \rmsg{\xAudit}{\xUser}{-19.55}{عرض + تصدير \EN{CSV}}
  % Self-loop on the Audit Log lifeline: the admin's access is recorded
  \draw[selfloop] (\xAudit - 0.15, -19.95)
    -- (\xAudit - 0.85, -19.95)
    -- (\xAudit - 0.85, -20.20)
    -- (\xAudit - 0.15, -20.20);
  \node[mlabel, left] at (\xAudit - 0.90, -20.07) {تسجيل وصول المسؤول};

  %==========================================================================
  %  UC-08: تشغيل التهيئة الذكية   (block: y = -20.5 .. -23.5)
  %    Onboarding Operator → Frontend → Smart Views + Qdrant indexing
  %    → readiness report
  %==========================================================================
  \draw[ucblock, fill=kauaccent!6] (-0.9, -20.5) rectangle (19.3, -23.5);
  \node[uclabel, anchor=north west] at (-0.85, -20.55)
    {\EN{UC-08:} تشغيل التهيئة الذكية};

  \smsg{\xUser}{\xFront}{-21.15}{بدء التهيئة الذكية}
  \smsg{\xFront}{\xPg}{-21.70}{بناء \EN{Smart Views}}
  \smsg{\xFront}{\xQd}{-22.25}{فهرسة المتجهات}
  \rmsg{\xFront}{\xUser}{-22.80}{تقرير الجاهزية}
  \smsg{\xFront}{\xAudit}{-23.35}{تسجيل التدقيق}

  %==========================================================================
  %  FOOTER HEADER BOXES  (repeat lifeline labels for readability)
  %==========================================================================
  \foreach \x/\lbl in {
    \xUser/{المستخدم المؤسسي\\\EN{(Actor)}},
    \xFront/{الواجهة الأمامية\\\EN{(Frontend)}},
    \xSup/{الوكيل المشرف\\\EN{(Supervisor)}},
    \xSql/{وكيل \EN{SQL}},
    \xRag/{وكيل \EN{RAG}},
    \xAna/{وكيل المحلل\\\EN{(Analyst)}},
    \xPg/{\EN{PostgreSQL}},
    \xQd/{\EN{Qdrant}},
    \xAudit/{سجل التدقيق\\\EN{(Audit Log)}}%
  } {
    \node[header] at (\x, -24.1) {\lbl};
  }

\end{tikzpicture}%
}
